```python
def _short_letters_hold(a, letter):
    """A single-dash word whose other characters are ASCII letters, one of them letter."""
    if len(a) < 2 or a[0] != "-" or a[1] == "-":
        return False
    has = False
    for ch in a[1:]:
        if not (("A" <= ch <= "Z") or ("a" <= ch <= "z")):
            return False
        if ch == letter:
            has = True
    return has
```
